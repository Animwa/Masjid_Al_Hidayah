const SPREADSHEET_ID = SpreadsheetApp.getActiveSpreadsheet().getId();

function doGet(e) {
  const action = e.parameter.action;
  let result = { success: false, message: "Aksi tidak dikenal" };

  try {
    initSheets(); // Pastikan struktur sheet tersedia

    if (action === "get_all_data") {
      result = {
        success: true,
        pengurus: getSheetData("Pengurus"),
        inventaris: getSheetData("Inventaris"),
        jamaah: getSheetData("Jamaah"),
        presensi: getSheetData("Presensi"),
        admins: getSheetData("Admins").map(a => ({ nama: a.Nama, role: a.Role })) // Sembunyikan PIN
      };
    } else if (action === "get_rekap_bulanan") {
      result = { success: true, data: getRekapBulanan() };
    }
  } catch (err) {
    result = { success: false, error: err.toString() };
  }

  return responseJSON(result);
}

function doPost(e) {
  let result = { success: false, message: "Aksi gagal" };
  try {
    initSheets();
    const payload = JSON.parse(e.postData.contents);
    const action = payload.action;

    if (action === "login") {
      result = verifyAdmin(payload.nama, payload.pin);
    } else if (action === "add_admin") {
      result = addAdmin(payload.nama, payload.pin, payload.adminSession);
    } else if (action === "save_pengurus") {
      result = saveData("Pengurus", ["ID", "Nama", "Jabatan", "NoHP", "Status"], payload.data);
    } else if (action === "delete_pengurus") {
      result = deleteData("Pengurus", payload.id);
    } else if (action === "save_inventaris") {
      result = saveData("Inventaris", ["ID", "NamaBarang", "Jumlah", "Kondisi", "TanggalMasuk", "Keterangan"], payload.data);
    } else if (action === "delete_inventaris") {
      result = deleteData("Inventaris", payload.id);
    } else if (action === "save_jamaah") {
      result = saveData("Jamaah", ["ID", "Nama", "TanggalLahir", "Gender", "Alamat", "Status"], payload.data);
    } else if (action === "delete_jamaah") {
      result = deleteData("Jamaah", payload.id);
    } else if (action === "save_presensi_batch") {
      result = savePresensiBatch(payload.records);
    }
  } catch (err) {
    result = { success: false, error: err.toString() };
  }

  return responseJSON(result);
}

function responseJSON(data) {
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

// Inisialisasi Sheet jika belum ada
function initSheets() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheets = {
    "Admins": ["Nama", "PIN", "Role", "CreatedAt"],
    "Pengurus": ["ID", "Nama", "Jabatan", "NoHP", "Status"],
    "Inventaris": ["ID", "NamaBarang", "Jumlah", "Kondisi", "TanggalMasuk", "Keterangan"],
    "Jamaah": ["ID", "Nama", "TanggalLahir", "Gender", "Alamat", "Status"],
    "Presensi": ["ID", "Kelompok", "Kelas", "Tanggal", "Hari", "NamaJamaah", "StatusPresensi", "Timestamp"]
  };

  for (let name in sheets) {
    let sheet = ss.getSheetByName(name);
    if (!sheet) {
      sheet = ss.insertSheet(name);
      sheet.appendRow(sheets[name]);
      if (name === "Admins") {
        // Default Admin Utama (Bisa diubah setelah deployment)
        sheet.appendRow(["Admin Utama", "123456", "Utama", new Date().toISOString()]);
      }
    }
  }
}

function getSheetData(sheetName) {
  const sheet = SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName(sheetName);
  if (!sheet) return [];
  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];
  const headers = data[0];
  return data.slice(1).map(row => {
    let obj = {};
    headers.forEach((h, i) => obj[h] = row[i]);
    return obj;
  });
}

function verifyAdmin(nama, pin) {
  const admins = getSheetData("Admins");
  const found = admins.find(a => String(a.Nama).toLowerCase() === String(nama).toLowerCase() && String(a.PIN) === String(pin));
  if (found) {
    return { success: true, admin: { nama: found.Nama, role: found.Role } };
  }
  return { success: false, message: "Nama atau PIN Admin salah!" };
}

function addAdmin(nama, pin, session) {
  // Validasi Admin Utama
  const auth = verifyAdmin(session.nama, session.pin);
  if (!auth.success || auth.admin.role !== "Utama") {
    return { success: false, message: "Hanya Admin Utama yang berhak menambah Admin baru!" };
  }
  const sheet = SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName("Admins");
  sheet.appendRow([nama, String(pin), "Pengurus", new Date().toISOString()]);
  return { success: true, message: "Admin baru berhasil ditambahkan!" };
}

function saveData(sheetName, headers, rowObj) {
  const sheet = SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName(sheetName);
  const data = sheet.getDataRange().getValues();
  let foundIndex = -1;

  if (!rowObj.ID) {
    rowObj.ID = "ID-" + new Date().getTime();
  }

  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]) === String(rowObj.ID)) {
      foundIndex = i + 1;
      break;
    }
  }

  const rowData = headers.map(h => rowObj[h] !== undefined ? rowObj[h] : "");

  if (foundIndex > 0) {
    sheet.getRange(foundIndex, 1, 1, headers.length).setValues([rowData]);
  } else {
    sheet.appendRow(rowData);
  }
  return { success: true, message: "Data berhasil disimpan!" };
}

function deleteData(sheetName, id) {
  const sheet = SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName(sheetName);
  const data = sheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]) === String(id)) {
      sheet.deleteRow(i + 1);
      return { success: true, message: "Data berhasil dihapus!" };
    }
  }
  return { success: false, message: "Data tidak ditemukan!" };
}

function savePresensiBatch(records) {
  const sheet = SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName("Presensi");
  const timestamp = new Date().toISOString();
  
  records.forEach(rec => {
    const id = "PRS-" + new Date().getTime() + "-" + Math.floor(Math.random()*1000);
    sheet.appendRow([id, rec.kelompok, rec.kelas, rec.tanggal, rec.hari, rec.nama, rec.status, timestamp]);
  });
  
  return { success: true, message: "Presensi berhasil disimpan!" };
}

function getRekapBulanan() {
  const presensi = getSheetData("Presensi");
  const now = new Date();
  const oneMonthAgo = new Date(now.getFullYear(), now.getMonth() - 1, now.getDate());

  let stats = {};

  presensi.forEach(p => {
    const pDate = new Date(p.Tanggal);
    if (pDate >= oneMonthAgo) {
      const key = `${p.Kelompok} - ${p.Kelas}`;
      if (!stats[key]) {
        stats[key] = { Hadir: 0, Izin: 0, Alfa: 0 };
      }
      if (stats[key][p.StatusPresensi] !== undefined) {
        stats[key][p.StatusPresensi]++;
      }
    }
  });

  return stats;
}
